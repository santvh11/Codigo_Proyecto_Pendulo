// =====================================================================
// FORMULA NODE 2 — Calculo de zeta y wn (va DESPUES del While Loop)
// ---------------------------------------------------------------------
// ENTRADAS:
//   runs     1D array I32  recorridos en cuentas (salida del ciclo)
//   tpk      1D array DBL  tiempos de cada extremo [s]
//   minRun   I32           recorridos menores a esto se ignoran (recomendado 20)
// SALIDAS:
//   zeta     DBL  factor de amortiguamiento
//   wn       DBL  frecuencia natural [rad/s]
//   Td       DBL  periodo amortiguado [s]
//   delta    DBL  decremento logaritmico (por ciclo completo)
//   valido   I32  1 si hubo datos suficientes, 0 si no
// Metodo: ajuste por minimos cuadrados de ln(run) vs indice k.
//   Cada recorrido es pico a pico, asi que la razon entre recorridos
//   seguidos es la de medio ciclo: delta = -2 * pendiente.
//   zeta = delta / sqrt(4*pi^2 + delta^2)
//   Td   = 2 * (tiempo promedio entre extremos seguidos)
//   wn   = (2*pi/Td) / sqrt(1 - zeta^2)
// =====================================================================

float64 sk, sy, skk, sky, sd, y, s;
int32 i, n, N, nd;

sk = 0; sy = 0; skk = 0; sky = 0; sd = 0;
N = 0; nd = 0;
n = sizeOfDim(runs, 0);

for (i = 0; i < n; i++) {
    if (runs[i] >= minRun) {
        y = ln(runs[i]);
        sk = sk + i;  sy = sy + y;  skk = skk + i*i;  sky = sky + i*y;
        N = N + 1;
        if (i > 0) {
            if (runs[i-1] >= minRun) {
                sd = sd + (tpk[i] - tpk[i-1]);
                nd = nd + 1;
            }
        }
    }
}

if (N >= 3 && nd >= 1) {
    s     = (N*sky - sk*sy) / (N*skk - sk*sk);
    delta = -2*s;
    zeta  = delta / sqrt(4*pi*pi + delta*delta);
    Td    = 2*sd/nd;
    wn    = (2*pi/Td) / sqrt(1 - zeta*zeta);
    valido = 1;
} else {
    delta = 0; zeta = 0; Td = 0; wn = 0;
    valido = 0;
}
